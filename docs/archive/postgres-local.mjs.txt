import EmbeddedPostgres from 'embedded-postgres';
import {existsSync,mkdirSync} from 'node:fs';
import {resolve} from 'node:path';
const url=new URL(process.env.DATABASE_URL);
if(url.hostname!=='127.0.0.1')throw new Error('Base locale uniquement.');
const directory=resolve('.data/postgres-native');mkdirSync(directory,{recursive:true});
const postgres=new EmbeddedPostgres({databaseDir:directory,user:decodeURIComponent(url.username),password:decodeURIComponent(url.password),port:Number(url.port||5432),persistent:true,authMethod:'scram-sha-256',postgresFlags:['-h','127.0.0.1','-k',directory]});
if(!existsSync(resolve(directory,'PG_VERSION')))await postgres.initialise();
await postgres.start();
console.log('PostgreSQL local démarré sur 127.0.0.1:'+url.port);
